# nvinfer config — SCRFD Face Detection (PGIE)
# Model: det_500m.onnx (buffalo_sc) đã re-export thành face_det_scrfd_b1.onnx
# với outputs 3-D [batch, anchors, K] qua scripts/fix_scrfd_onnx.py.
# Input: 640x640 RGB, normalize (x-127.5)/128 ≈ scale 1/128, mean 127.5
# Output: 9 tensors (3 strides × {score, bbox, kps}). Parser emit bbox sau
#         khi lọc score + hình học landmark; nvdsfaceembed decode lại
#         landmarks từ frame tensor meta để align + tạo embedding.

[property]
gpu-id=0

# Normalize: SCRFD dùng (x - 127.5) / 128.0
# nvinfer formula: net-scale-factor * (input - offsets)
net-scale-factor=0.0078125
offsets=127.5;127.5;127.5
model-color-format=0
# 0=RGB, 1=BGR. SCRFD ONNX expects RGB.

model-engine-file=/home/somethink/parking_system/models/face_det_scrfd_b1_fp16.engine
onnx-file=/home/somethink/parking_system/models/face_det_scrfd_b1.onnx

batch-size=2
# 2 sources, nhưng face detection chỉ cần chạy meaningful trên src1.
# nvinfer không support per-source batch trong DeepStream 7.1 mà không
# nvstreamdemux/nvstreammux split — chấp nhận chạy thừa trên src0 (~3ms).

network-mode=2
# 2=FP16

num-detected-classes=1
# 1 class: face

interval=0
gie-unique-id=2
process-mode=1
# 1=primary (chạy trên full frame, không dependent vào uid khác)

network-type=0
# 0=detector — parser trả NvDsInferParseObjectInfo (bbox), nvinfer auto-
# transform sang frame coords (1280x720) qua maintain-aspect-ratio.
# Landmarks decode trong nvdsfaceembed từ raw tensor (xem
# output-tensor-meta=1 dưới).

cluster-mode=4
# 4=NONE — parser tự làm NMS

maintain-aspect-ratio=1
symmetric-padding=1
infer-dims=3;640;640
# CHW format

# Required: nvdsfaceembed đọc 9 raw tensors từ frame_user_meta.
output-tensor-meta=1
parse-bbox-func-name=NvDsInferParseCustomScrfd
custom-lib-path=/home/somethink/parking_system/parsers/libnvds_scrfd_parser.so

# operate-on-class-ids không apply cho primary detector
# operate-on-gie-id không set vì process-mode=1

[class-attrs-all]
# Parser lấy ngưỡng này, sau đó lọc thêm hình học landmark và NMS.
# 0.50 giữ recall tốt hơn; chặn false positive ở downstream bằng yêu cầu
# phải có landmark + embedding trước khi publish lên dashboard/main.
pre-cluster-threshold=0.50
nms-iou-threshold=1.0
